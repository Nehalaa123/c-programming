  1  #include<stdio.h>
  2 #include<stdlib.h>
  3 struct Node{
  4     int data;
  5     struct Node *next;
  6 };
  7 struct Node *front = NULL;
  8 struct Node *rear = NULL;
  9 void enqueue(int value)
 10 {
 11     struct Node *newNode = (struct Node*)malloc(sizeof(struct Node));
 12     if (newNode == NULL)
 13     {
 14         printf("Queue overflow\n");
 15         return;
 16     }
 17     newNode->data = value;
 18     newNode->next = NULL;
 19     if(rear == NULL)
 20     {
 21         front = rear = newNode;
 22     }
 23     else{
 24         rear->next=newNode;
 25         rear=newNode;
 26     }
 27     printf("%d enqueued to queue\n",value);
 28 }
 29 void dequeue()
 30 {
 31     if(front == NULL)
 32     {
 33         printf("Queue underflow\n");
 34         return;
 35     }
 36     struct Node *temp = front;
 37     printf("%d dequeued from queue\n",front->data);
 38     front = front->next;
 39     if(front == NULL)
 40         rear = NULL;
 41     free(temp);
 42 }
 43 void display()
 44 {
 45     if(front == NULL)
 46  {
 47         printf("Queue is empty\n");
 48         return;
 49     }
 50     struct Node *temp = front;
 51     printf("Queue elements:");
 52     while(temp!= NULL)
 53     {
 54         printf("%d ->",temp->data);
 55         temp = temp->next;
 56     }
 57     printf("NULL\n");
 58 }
